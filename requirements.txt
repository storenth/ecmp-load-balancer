#   python3.11 -m pip install -r requirements.txt

#   python3 -m venv venv
#   source venv/bin/activate  # On Windows: venv\Scripts\activate
#   pip install -r requirements.txt

#   pipenv install

# =============================================================================
# Troubleshooting
# =============================================================================
#
# If you encounter installation issues:
#
# 1. Upgrade pip first:
#    pip install --upgrade pip
#
# 2. Install packages individually to identify problematic ones:
#    pip install pyyaml
#    pip install jinja2
#    pip install numpy
#    pip install scipy
#    pip install matplotlib
#    pip install pandas
#
# 3. Use pre-built wheels for faster installation:
#    pip install --only-binary :all: -r requirements.txt
#
# 4. If building from source fails, install build dependencies:
#    pip install wheel setuptools
#
# 5. For macOS, you may need to install system dependencies:
#    brew install python3
#
# 6. For Linux, you may need to install system dependencies:
#    sudo apt-get install python3-dev python3-pip
#    sudo apt-get install libyaml-dev
